   119	        public void LootItem(Item item, AutoLootConfigEntry entry = null)
   120	        {
   121	            if (item == null || !_recentlyLooted.Add(item.Serial) || !_quickContainsLookup.Add(item.Serial)) return;
   122	
   123	            // Skip corpse/container items that are on cooldown from a previous loot attempt.
   124	            // Ground items are excluded — their retry cadence is already gated by position changes.
   125	            if (!item.OnGround && _lootCooldowns.TryGetValue(item.Serial, out long cooldownUntil) && Time.Ticks < cooldownUntil)
   126	            {
   127	                _recentlyLooted.Remove(item.Serial);
   128	                _quickContainsLookup.Remove(item.Serial);
   129	                return;
   130	            }
   131	
   132	            int pri = entry != null ? -(int)entry.Priority : -(int)AutoLootPriority.Normal;
   133	            // Corpse items get a priority boost so they're dequeued before ground items
   134	            // of the same AutoLootPriority. This is handled here (not in the ObjectActionQueue)
   135	            // to avoid OAQ priority starvation where ground items would never be processed
   136	            // while corpse items keep flowing in.
   137	            if (!item.OnGround) pri -= 100;
   138	            _lootItems.Enqueue((item.Serial, entry), pri);
   139	            _currentLootTotalCount++;
   140	            _nextClearRecents = Time.Ticks + 5000;
   141	        }

   157	        /// <summary>
   158	        /// Check an item against the loot list, if it needs to be auto looted it will be.
   159	        /// </summary>
   160	        private void CheckAndLoot(Item i)
   161	        {
   162	            if (!_loaded || i == null || _quickContainsLookup.Contains(i.Serial)) return;
   163	
   164	            if(i.IsCorpse)
   165	            {
   166	                HandleCorpse(i);
   167	
   168	                return;
   169	            }
   170	
   171	            if (i.ShouldAutoLoot)
   172	            {
   173	                LootItem(i, null);
   174	                return;
   175	            }
   176	
   177	            AutoLootConfigEntry entry = IsOnLootList(i);
   178	            if (entry != null) LootItem(i, entry);
   179	        }

   181	        /// <summary>
   182	        /// Check if an item is on the auto loot list.
   183	        /// Uses graphic index for O(1) lookup by graphic ID, with match cache for repeated checks.
   184	        /// Multiple active profiles: returns the match with highest Priority.
   185	        /// </summary>
   186	        /// <param name="i">The item to check the loot list against</param>
   187	        /// <returns>The matched AutoLootConfigEntry, or null if no match found</returns>
   188	        internal AutoLootConfigEntry IsOnLootList(Item i)
   189	        {
   190	            if (!_loaded) return null;
   191	
   192	            // Check global exclusion list before profile matching
   193	            if (IsExcluded(i)) return null;
   194	
   195	            bool oplChecked = false;
   196	            bool oplAvailable = false;
   197	
   198	            // Check cache first
   199	            if (_matchCache.TryGetValue(i.Serial, out AutoLootConfigEntry cached))
   200	            {
   201	                if (cached != null)
   202	                    return cached; // Positive cache hit — already matched
   203	
   204	                // Negative cache hit — check for stale OPL state
   205	                oplAvailable = _world.OPL.TryGetNameAndData(i.Serial, out _, out _);
   206	                oplChecked = true;
   207	                bool hadOplWhenCached = _matchCacheHasOpl.Contains(i.Serial);
   208	
   209	                if (!oplAvailable || hadOplWhenCached)
   210	                    return null; // Valid negative cache
   211	
   212	                // OPL is now available but wasn't when we cached — stale, fall through to re-evaluate
   213	            }
   214	
   215	            // Cache miss or stale negative cache — run matching logic
   216	            if (!oplChecked)
   217	                oplAvailable = _world.OPL.TryGetNameAndData(i.Serial, out _, out _);
   218	            AutoLootConfigEntry result = null;
   219	
   220	            bool isGroundItem = i.OnGround;
   221	
   222	            if (_graphicIndex.TryGetValue(i.Graphic, out List<AutoLootConfigEntry> entries))
   223	                foreach (AutoLootConfigEntry entry in entries)
   224	                {
   225	                    if (isGroundItem && !entry.Scavenge) continue;
   226	                    if (entry.Match(i))
   227	                    {
   228	                        if (result == null || entry.Priority > result.Priority)
   229	                            result = entry;
   230	                        if (result.Priority == AutoLootPriority.High)
   231	                            break;
   232	                    }
   233	                }
   234	
   235	            if (result == null || result.Priority != AutoLootPriority.High)
   236	                foreach (AutoLootConfigEntry entry in _wildcardEntries)
   237	                {
   238	                    if (isGroundItem && !entry.Scavenge) continue;
   239	                    if (entry.Match(i))
   240	                    {
   241	                        if (result == null || entry.Priority > result.Priority)
   242	                            result = entry;
   243	                        if (result.Priority == AutoLootPriority.High)
   244	                            break;
   245	                    }
   246	                }
   247	
   248	            // Store result in cache
   249	            _matchCache[i.Serial] = result;
   250	
   251	            if (oplAvailable)
   252	                _matchCacheHasOpl.Add(i.Serial);
   253	            else
   254	                _matchCacheHasOpl.Remove(i.Serial);
   255	
   256	            return result;
   257	        }

   259	        /// <summary>
   260	        /// Add an entry for auto looting to match against when opening corpses.
   261	        /// Routes to SelectedProfile's entry list.
   262	        /// </summary>
   263	        /// <param name="graphic"></param>
   264	        /// <param name="hue"></param>
   265	        /// <param name="name"></param>
   266	        /// <returns></returns>
   267	        public AutoLootConfigEntry AddAutoLootEntry(ushort graphic = 0, ushort hue = ushort.MaxValue, string name = "")
   268	        {
   269	            AutoLootProfile profile = EnsureSelectedProfile();
   270	            if (profile == null)
   271	                return null;
   272	
   273	            var item = new AutoLootConfigEntry() { Graphic = graphic, Hue = hue, Name = name };
   274	
   275	            foreach (AutoLootConfigEntry entry in profile.Entries)
   276	                if (entry.Equals(item))
   277	                    return entry;
   278	
   279	            profile.Entries.Add(item);
   280	            RebuildMergedList();
   281	            SaveProfile(profile);
   282	
   283	            return item;
   284	        }

   286	        private AutoLootProfile EnsureSelectedProfile()
   287	        {
   288	            if (SelectedProfile != null)
   289	                return SelectedProfile;
   290	
   291	            if (!_loaded)
   292	            {
   293	                Log.Warn("AddAutoLootEntry called before profiles are loaded");
   294	                return null;
   295	            }
   296	
   297	            if (Profiles.Count > 0)
   298	            {
   299	                SelectedProfile = Profiles[0];
   300	                return SelectedProfile;
   301	            }
   302	
   303	            SelectedProfile = CreateProfile("Default");
   304	            return SelectedProfile;
   305	        }

   307	        /// <summary>
   308	        /// Search through a corpse and check items that need to be looted.
   309	        /// Only call this after checking that autoloot IsEnabled
   310	        /// Note: This method doesn't gurantee to process all itmes in the corpse,
   311	        /// because `corpse.Items` is populated via `AddItemToContainer` packet, thus
   312	        /// it may not have all items yet when the method is called.
   313	        /// </summary>
   314	        /// <param name="corpse"></param>
   315	        private void HandleCorpse(Item corpse)
   316	        {
   317	            if (corpse is not { IsCorpse: true }) return;
   318	
   319	            if (corpse.Distance > ProfileManager.CurrentProfile.AutoOpenCorpseRange)
   320	            {
   321	                World.Instance?.Player?.AutoOpenedCorpses.Remove(corpse); //Retry if the distance was too great to loot
   322	                return;
   323	            }
   324	
   325	            if (corpse.IsHumanCorpse && !ProfileManager.CurrentProfile.AutoLootHumanCorpses) return;
   326	
   327	            for (LinkedObject i = corpse.Items; i != null; i = i.Next)
   328	                CheckAndLoot((Item)i);
   329	
   330	            if(ProfileManager.CurrentProfile.HueCorpseAfterAutoloot)
   331	                corpse.Hue = 73;
   332	        }

   334	        public void TryRemoveAutoLootEntry(string uid)
   335	        {
   336	            if (!_loaded)
   337	                return;
   338	
   339	            AutoLootProfile profile = SelectedProfile;
   340	            if (profile == null)
   341	                return;
   342	
   343	            int removeAt = -1;
   344	
   345	            for (int i = 0; i < profile.Entries.Count; i++)
   346	            {
   347	                if (profile.Entries[i].Uid == uid)
   348	                {
   349	                    removeAt = i;
   350	                    break;
   351	                }
   352	            }
   353	
   354	            if (removeAt > -1)
   355	            {
   356	                profile.Entries.RemoveAt(removeAt);
   357	                RebuildMergedList();
   358	                SaveProfile(profile);
   359	            }
   360	        }

   362	        /// <summary>
   363	        /// Checks if item is a corpse, or if its root container is corpse and handles them appropriately.
   364	        /// </summary>
   365	        /// <param name="i"></param>
   366	        private void CheckCorpse(Item i)
   367	        {
   368	            if (i == null) return;
   369	
   370	            if (i.IsCorpse)
   371	            {
   372	                HandleCorpse(i);
   373	                return;
   374	            }
   375	
   376	            Item root = _world.Items.Get(i.RootContainer);
   377	            if (root != null && root.IsCorpse)
   378	            {
   379	                // Check the item that triggered this call directly
   380	                CheckAndLoot(i);
   381	                // A defensive safety net to ensure all items in the corpse are processed
   382	                HandleCorpse(root);
   383	                return;
   384	            }
   385	        }

   417	        internal void OnPositionChanged(object sender, PositionChangedArgs e)
   418	        {
   419	            if (!_loaded) return;
   420	
   421	            if (ProfileManager.CurrentProfile.EnableScavenger)
   422	            {
   423	                if (_nearbyGroundItems.Count == 0)
   424	                {
   425	                    BootstrapNearbyGroundItems();
   426	                    return;
   427	                }
   428	
   429	                List<uint> toRemove = null;
   430	                List<Item> toLoot = null;
   431	
   432	                // Snapshot to avoid collection-modified-during-enumeration if CheckAndLoot
   433	                // triggers events that call OnItemCreatedOrUpdated (which modifies _nearbyGroundItems).
   434	                int count = _nearbyGroundItems.Count;
   435	                if (_nearbyItemsSnapshot.Length < count)
   436	                    _nearbyItemsSnapshot = new uint[count];
   437	                _nearbyGroundItems.CopyTo(_nearbyItemsSnapshot);
   438	
   439	                for (int idx = 0; idx < count; idx++)
   440	                {
   441	                    uint serial = _nearbyItemsSnapshot[idx];
   442	                    Item item = _world.Items.Get(serial);
   443	
   444	                    if (item == null || !item.OnGround || item.IsLocked || item.IsCorpse)
   445	                    {
   446	                        (toRemove ??= new List<uint>()).Add(serial);
   447	                        continue;
   448	                    }
   449	
   450	                    if (item.Distance >= SCAVENGER_TRACKING_RADIUS)
   451	                    {
   452	                        (toRemove ??= new List<uint>()).Add(serial);
   453	                        continue;
   454	                    }
   455	
   456	                    if (item.Distance <= ProfileManager.CurrentProfile.AutoOpenCorpseRange)
   457	                        (toLoot ??= new List<Item>()).Add(item);
   458	                }
   459	
   460	                if (toRemove != null)
   461	                    foreach (uint serial in toRemove)
   462	                        _nearbyGroundItems.Remove(serial);
   463	
   464	                // Loot after iteration is complete to avoid re-entrancy issues
   465	                if (toLoot != null)
   466	                    foreach (Item item in toLoot)
   467	                        CheckAndLoot(item);
   468	            }
   469	
   470	            if (IsEnabled)
   471	                foreach (Item corpse in _world.GetCorpseSnapshot())
   472	                    CheckCorpse(corpse);
   473	        }

   482	        private void OnItemCreatedOrUpdated(object sender, EventArgs e)
   483	        {
   484	            if (!_loaded) return;
   485	
   486	            if (sender is Item i)
   487	            {
   488	                _matchCache.Remove(i.Serial);
   489	                _matchCacheHasOpl.Remove(i.Serial);
   490	
   491	                if (IsEnabled)
   492	                    CheckCorpse(i);
   493	
   494	                // Maintain spatial tracking set for scavenger and attempt immediate pickup.
   495	                if (ProfileManager.CurrentProfile.EnableScavenger)
   496	                {
   497	                    if (IsTrackableGroundItem(i) && i.Distance <= SCAVENGER_TRACKING_RADIUS)
   498	                    {
   499	                        _nearbyGroundItems.Add(i.Serial);
   500	
   501	                        // Attempt immediate scavenge for items within pickup range
   502	                        if (i.Distance <= ProfileManager.CurrentProfile.AutoOpenCorpseRange)
   503	                            CheckAndLoot(i);
   504	                    }
   505	                    else
   506	                    {
   507	                        _nearbyGroundItems.Remove(i.Serial);
   508	                    }
   509	                }
   510	            }
   511	        }

   513	        internal void BootstrapNearbyGroundItems()
   514	        {
   515	            _nearbyGroundItems.Clear();
   516	
   517	            foreach (Item item in _world.Items.Values)
   518	            {
   519	                if (IsTrackableGroundItem(item) && item.Distance <= SCAVENGER_TRACKING_RADIUS)
   520	                    _nearbyGroundItems.Add(item.Serial);
   521	            }
   522	        }

   524	        private void OnOPLReceived(object sender, OPLEventArgs e)
   525	        {
   526	            if (!_loaded || !IsEnabled) return;
   527	            _matchCache.Remove(e.Serial);
   528	            _matchCacheHasOpl.Remove(e.Serial);

   534	        public void Update()
   535	        {
   536	            if (!_loaded || (!IsEnabled && !ProfileManager.CurrentProfile.EnableScavenger) || !_world.InGame) return;
   537	
   538	            if (_nextLootTime > Time.Ticks) return;
   539	
   540	            if (Client.Game.UO.GameCursor.ItemHold.Enabled)
   541	                return; //Prevent moving stuff while holding an item.
   542	
   543	            // Periodically clear _recentlyLooted so ground items can be retried
   544	            // even while the queue is active (e.g., corpse looting in progress).
   545	            if (Time.Ticks > _nextClearRecents)
   546	            {
   547	                _recentlyLooted.Clear();
   548	                ClearMatchCache();
   549	                _nextClearRecents = Time.Ticks + 5000;
   550	
   551	                // Clean up expired cooldowns and cooldowns for items that no longer exist
   552	                List<uint> expiredCooldowns = null;
   553	                foreach (var kvp in _lootCooldowns)
   554	                {
   555	                    if (Time.Ticks >= kvp.Value || _world.Items.Get(kvp.Key) == null)
   556	                        (expiredCooldowns ??= new List<uint>()).Add(kvp.Key);
   557	                }
   558	
   559	                if (expiredCooldowns != null)
   560	                    foreach (uint serial in expiredCooldowns)
   561	                        _lootCooldowns.Remove(serial);
   562	            }
   563	
   564	            if (_lootItems.Count == 0)
   565	            {
   566	                _progressBarGump?.Dispose();
   567	                return;
   568	            }
   569	
   570	            (uint item, AutoLootConfigEntry entry) = _lootItems.Dequeue();
   571	            if (item == 0) return;
   572	
   573	            if (_lootItems.Count == 0) //Que emptied out
   574	                _currentLootTotalCount = 0;
   575	
   576	            _quickContainsLookup.Remove(item);
   577	
   578	            Item moveItem = _world.Items.Get(item);
   579	
   580	            if (moveItem == null)
   581	                return;
   582	
   583	            CreateProgressBar();
   584	
   585	            if (_progressBarGump is { IsDisposed: false }) _progressBarGump.CurrentPercentage = 1 - ((double)_lootItems.Count / (double)_currentLootTotalCount);
   586	
   587	            if (moveItem.Distance > ProfileManager.CurrentProfile.AutoOpenCorpseRange)
   588	            {
   589	                Item rc = _world.Items.Get(moveItem.RootContainer);
   590	                if (rc != null && rc.Distance > ProfileManager.CurrentProfile.AutoOpenCorpseRange)
   591	                {
   592	                    if (rc.IsCorpse)
   593	                        World.Instance?.Player?.AutoOpenedCorpses.Remove(rc); //Allow reopening this corpse, we got too far away to finish looting..
   594	                    _recentlyLooted.Remove(item);
   595	                    return;
   596	                }
   597	            }
   598	
   599	            uint destinationSerial = 0;
   600	
   601	            //If this entry has a specific container, use it
   602	            if (entry != null && entry.DestinationContainer != 0)
   603	            {
   604	                Item itemDestContainer = _world.Items.Get(entry.DestinationContainer);
   605	                if (itemDestContainer != null) destinationSerial = entry.DestinationContainer;
   606	            }
   607	
   608	            //If the entry's profile has a default container, use it
   609	            if (destinationSerial == 0 && entry != null && entry.ProfileDestinationContainer != 0)
   610	            {
   611	                Item profileDestContainer = _world.Items.Get(entry.ProfileDestinationContainer);
   612	                if (profileDestContainer != null) destinationSerial = entry.ProfileDestinationContainer;
   613	            }
   614	
   615	            if (destinationSerial == 0 && ProfileManager.CurrentProfile.GrabBagSerial != 0)
   616	            {
   617	                Item grabBag = _world.Items.Get(ProfileManager.CurrentProfile.GrabBagSerial);
   618	                if (grabBag != null) destinationSerial = ProfileManager.CurrentProfile.GrabBagSerial;
   619	            }
   620	
   621	            if (destinationSerial == 0)
   622	            {
   623	                Item backpack = _world.Player.Backpack;
   624	                if (backpack != null) destinationSerial = backpack.Serial;
   625	            }
   626	
   627	            if (destinationSerial != 0)
   628	            {
   629	                ActionPriority lootPriority = entry?.Priority switch
   630	                {
   631	                    AutoLootPriority.High => ActionPriority.LootItemHigh,
   632	                    AutoLootPriority.Low => ActionPriority.LootItem,
   633	                    _ => ActionPriority.LootItemMedium,
   634	                };
   635	                ObjectActionQueue.Instance.Enqueue(new MoveRequest(moveItem.Serial, destinationSerial, moveItem.Amount).ToObjectActionQueueItem(), lootPriority);
   636	
   637	                // Set a 30-second cooldown for corpse/container items so if the move fails, we don't spam retry.
   638	                // Ground items don't need this — their retry is already gated by position changes.
   639	                if (!moveItem.OnGround)
   640	                    _lootCooldowns[moveItem.Serial] = Time.Ticks + 30000;
   641	            }
   642	            else
   643	                GameActions.Print("Could not find a container to loot into. Try setting a grab bag.");
   644	
   645	            _nextLootTime = Time.Ticks + ProfileManager.CurrentProfile.MoveMultiObjectDelay;
   646	        }

   662	        private void LoadProfiles()
   663	        {
   664	            Task.Factory.StartNew(() =>
   665	            {
   666	                try
   667	                {
   668	                    Directory.CreateDirectory(_profilesDir);
   669	                }
   670	                catch (Exception e)
   671	                {
   672	                    Log.Error($"Failed to create auto loot profiles directory: {e.Message}");
   673	                    return;
   674	                }
   675	
   676	                // Migration state detection: check conditions in priority order
   677	                string[] files;
   678	                try
   679	                {
   680	                    files = Directory.GetFiles(_profilesDir, "*.json");
   681	                }
   682	                catch (Exception e)
   683	                {
   684	                    Log.Error($"Failed to scan auto loot profiles directory: {e.Message}");
   685	                    return;
   686	                }
   687	
   688	                if (files.Length > 0)
   689	                {
   690	                    // State 1: Directory has .json files — normal load, no migration
   691	                    Log.Info("AutoLoot profiles directory found with existing profiles, loading normally.");
   692	                }
   693	                else
   694	                {
   695	                    // Directory is empty (or only has .backup files).
   696	                    // Check for legacy AutoLoot.json files to migrate.
   697	                    string ancientPath = Path.Combine(CUOEnviroment.ExecutablePath, "Data", "Profiles", "AutoLoot.json");
   698	
   699	                    if (File.Exists(_savePath))
   700	                    {
   701	                        // State 3: AutoLoot.json at ProfileManager.ProfilePath — flag for migration
   702	                        _migrationSourcePath = _savePath;
   703	                        Log.Info($"Legacy AutoLoot.json found at profile path, flagged for migration: {_savePath}");
   704	
   705	                        if (File.Exists(ancientPath))
   706	                            Log.Warn($"Legacy AutoLoot.json also exists at ancient path ({ancientPath}), using profile path version.");
   707	                    }
   708	                    else if (File.Exists(ancientPath))
   709	                    {
   710	                        // State 4: AutoLoot.json at ancient executable path — READ only, flag for migration
   711	                        _migrationSourcePath = ancientPath;
   712	                        Log.Info($"Legacy AutoLoot.json found at ancient path, flagged for migration: {ancientPath}");
   713	                    }
   714	                    else
   715	                    {
   716	                        // State 2/5: No legacy files exist — create empty Default profile
   717	                        Log.Info("No existing auto loot data found, creating Default profile.");
   718	                        var defaultProfile = CreateDefaultProfile();
   719	                        SaveProfile(defaultProfile);
   720	                        files = new[] { Path.Combine(_profilesDir, defaultProfile.FileName) };
   721	                    }
   722	                }
   723	
   724	                // Perform migration if a legacy file was detected
   725	                if (_migrationSourcePath != null)
   726	                {
   727	                    files = MigrateToDefaultProfile(files);
   728	                }
   729	                var loadedProfiles = new List<AutoLootProfile>();
   730	
   731	                foreach (string file in files)
   732	                {
   733	                    try
   734	                    {
   735	                        string json = File.ReadAllText(file);
   736	                        AutoLootProfile profile = JsonSerializer.Deserialize(json, AutoLootJsonContext.Default.AutoLootProfile);
   737	
   738	                        if (profile == null)
   739	                        {
   740	                            Log.Error($"Failed to deserialize auto loot profile: {file}");
   741	                            continue;
   742	                        }
   743	
   744	                        profile.FileName = Path.GetFileName(file);
   745	                        loadedProfiles.Add(profile);
   746	                    }
   747	                    catch (Exception e)
   748	                    {
   749	                        Log.Error($"Error loading auto loot profile '{file}': {e.Message}");
   750	                    }
   751	                }
   752	
   753	                loadedProfiles.Sort((a, b) => a.DisplayOrder.CompareTo(b.DisplayOrder));
   754	                Profiles = loadedProfiles;
   755	                // Build merged list before marking as loaded so other threads
   756	                // see a populated _mergedEntries when they check _loaded.
   757	                var newList = new List<AutoLootConfigEntry>();
   758	                int activeCount = 0;
   759	                foreach (AutoLootProfile profile in loadedProfiles)
   760	                {
   761	                    if (profile.IsActive)
   762	                    {
   763	                        newList.AddRange(profile.Entries);
   764	                        activeCount++;
   765	                    }
   766	                }
   767	                _activeProfileCount = activeCount;
   768	                _mergedEntries = newList;
   769	                RebuildGraphicIndex();
   770	                _loaded = true;
   771	            });
   772	        }

   926

  1224	            var newItems = new List<AutoLootConfigEntry>();

  1439	                return fileName;
